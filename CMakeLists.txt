cmake_minimum_required(VERSION 3.25)
project(OBSReplay VERSION 0.1.0 LANGUAGES CXX RC)

if(NOT WIN32)
  message(FATAL_ERROR "OBSReplay currently supports Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
set(FETCHCONTENT_UPDATES_DISCONNECTED ON CACHE BOOL "Do not update pinned dependencies during configuration" FORCE)
set(saucer_static ON CACHE BOOL "" FORCE)
set(saucer_examples OFF CACHE BOOL "" FORCE)
set(saucer_tests OFF CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
FetchContent_Declare(saucer GIT_REPOSITORY https://github.com/saucer/saucer GIT_TAG v8.0.5 GIT_SHALLOW TRUE)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(saucer nlohmann_json)

set(notices "OBSReplay 0.1.0\n\nThird-party software notices\n\n")
file(GLOB dependency_dirs LIST_DIRECTORIES true "${FETCHCONTENT_BASE_DIR}/*-src")
foreach(dependency_dir IN LISTS dependency_dirs)
  get_filename_component(dependency_name "${dependency_dir}" NAME)
  file(GLOB license_files "${dependency_dir}/LICENSE*" "${dependency_dir}/COPYING*")
  foreach(license_file IN LISTS license_files)
    if(NOT IS_DIRECTORY "${license_file}")
      file(READ "${license_file}" license_text)
      string(APPEND notices "\n======== ${dependency_name} ========\n${license_text}\n")
    endif()
  endforeach()
endforeach()
file(GLOB sdk_licenses "${FETCHCONTENT_BASE_DIR}/saucer-build/nuget/packages/Microsoft.Web.WebView2.*/LICENSE*")
foreach(sdk_license IN LISTS sdk_licenses)
  file(READ "${sdk_license}" license_text)
  string(APPEND notices "\n======== Microsoft WebView2 SDK / static loader ========\n${license_text}\n")
endforeach()
set(inter_license "${CMAKE_CURRENT_SOURCE_DIR}/frontend/fonts/OFL.txt")
if(EXISTS "${inter_license}")
  file(READ "${inter_license}" license_text)
  string(APPEND notices "\n======== Inter font ========\n${license_text}\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${inter_license}")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/THIRD-PARTY-NOTICES.txt" "${notices}")
file(TO_CMAKE_PATH "${CMAKE_CURRENT_BINARY_DIR}/THIRD-PARTY-NOTICES.txt" NOTICES_PATH)
configure_file(resources/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)

add_executable(OBSReplay WIN32
  src/main.cpp src/platform.cpp src/integration.cpp src/controller.cpp src/obs_socket.cpp
  "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
target_compile_definitions(OBSReplay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(OBSReplay PRIVATE /W4 /utf-8 /permissive- /EHsc)
target_link_libraries(OBSReplay PRIVATE saucer::saucer saucer::private saucer::embedded nlohmann_json::nlohmann_json
  comctl32 comdlg32 shell32 ole32 uuid propsys advapi32 bcrypt crypt32 winmm winhttp version dxgi)
saucer_embed("frontend" TARGET OBSReplay DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/embedded")
set_target_properties(OBSReplay PROPERTIES INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
target_link_options(OBSReplay PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup /MANIFEST:NO /OPT:REF /OPT:ICF)
install(TARGETS OBSReplay RUNTIME DESTINATION .)
